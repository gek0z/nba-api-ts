```ts
import { NBAClient } from "nba-api-ts";

const nba = new NBAClient({ stats: { fetch: browserFetch } });

const { careerTotalsRegularSeason } = await nba.stats.playerCareerStats({ playerID: 2544 });
console.log(careerTotalsRegularSeason[0].pts); // typed: number
```
